# ─── OGAP v2 Python requirements ──────────────────────────────────────────────
# Pinned to match the Compute-Canada Rorqual wheelhouse where possible. Install
# via setup_ogap_env_v2.sh which routes most lines to `pip install --no-index`
# (CC wheelhouse) and falls back to PyPI for items not in the local mirror.
#
# This file is for reference / reproducibility; the canonical installer is
# setup_ogap_env_v2.sh.
#
# REPRODUCIBILITY: after a successful publication run, freeze the EXACT resolved
# environment with `pip freeze > requirements_frozen.txt` and cite that in the
# Methods. The bounds below pin only the numerically load-bearing packages (a
# scipy/numpy major bump can change distance_transform_edt → HD95, or break the
# numpy-1.x ABI the scipy-stack 2024a wheelhouse + the sbatch preflight assume).

# ── Optional MedIA/IEEE TMI rigor extensions ────────────────────────────────
# These do not block training / evaluation if missing; they only disable a
# specific subcommand or cross-check. The setup script installs them with
# --no-deps so PyPI cannot shadow scipy-stack numpy or the pinned CC torch.

medpy            # metric_check medpy cross-validation (PyPI fallback)
highdicom        # dicom_seg subcommand (PyPI only - not in CC wheelhouse)

# ── CUDA-built optional dependency (PyPI source build) ──────────────────────
# The default installer skips these because current CC/PyPI metadata can pull
# mutually incompatible torch pins. Use setup_ogap_env_v2_mamba.sh to create a
# separate torch-2.5 SegMamba env instead of mixing these into ogap_env_v2.
# These two require nvcc + ninja and a working CUDA_HOME.
causal-conv1d    # SegMamba transitive dependency
mamba-ssm        # SegMambaTeacher backbone

# ── v9.1 ODE / NAS additions (optional, PyPI) ─────────────────────────

# torchdiffeq is required for true O(1)-memory adjoint training of the
# continuous-depth ODE teacher (ogap.models.ode, --teacher_arch ode). Fixed-step
# ODE ablations must set adjoint=False explicitly.
torchdiffeq      # required for --teacher_arch ode
